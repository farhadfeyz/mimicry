from hazm import Normalizer, word_tokenize, Stemmer, stopwords_list

def preprocess(text):
    text = normalizer.normalize(text)                     # normalizes Persian characters
    tokens = word_tokenize(text)                          # tokenizes
    tokens = [t for t in tokens if t not in persian_stopwords]  # remove stopwords
    tokens = [t for t in tokens if len(t) > 1]            # remove very short tokens
    # tokens = [stemmer.stem(t) for t in tokens]          # uncomment if you want stemming
    return tokens

import numpy as np

# Topic-word probability matrix: shape (num_topics, vocab_size)
phi = final_model.get_topics()  # each row is a topic, columns = id2word

# Vocabulary size
vocab_size = len(id2word)
num_topics = phi.shape[0]

# Compute overall word frequencies p(w) from the corpus
# We'll sum over all documents and then normalise
corpus_word_counts = np.zeros(vocab_size)
for doc_bow in corpus:
    for word_id, count in doc_bow:
        corpus_word_counts[word_id] += count

p_w = corpus_word_counts / corpus_word_counts.sum()

# Compute exclusivity for each (topic, word)
# excl[t,k] = phi[t,k] / sum over all topics of phi[:,k]
excl = np.zeros_like(phi)
for k in range(vocab_size):
    col_sum = phi[:, k].sum()
    if col_sum > 0:
        excl[:, k] = phi[:, k] / col_sum

# Compute ranks for frequency (within topic) and exclusivity (within topic)
freq_ranks = np.zeros_like(phi)
excl_ranks = np.zeros_like(phi)

for t in range(num_topics):
    # Rank frequency: highest probability -> rank 1
    freq_ranks[t, :] = np.argsort(np.argsort(-phi[t, :])) + 1
    # Rank exclusivity: highest exclusivity -> rank 1
    excl_ranks[t, :] = np.argsort(np.argsort(-excl[t, :])) + 1

# FREX score (harmonic mean of the two ranks, lower is better)
frex_score = 2.0 / (1.0 / freq_ranks + 1.0 / excl_ranks)

lam = 0.6  # you can tune this

# log p(w|k)  and  log p(w|k)/p(w)  = log(p(w|k)) - log(p(w))
log_phi = np.log(phi + 1e-12)          # avoid log(0)
log_p_w = np.log(p_w + 1e-12)

log_lift = log_phi - log_p_w[np.newaxis, :]   # log( p(w|k)/p(w) )

relevance = lam * log_phi + (1 - lam) * log_lift
# Higher relevance = better word for that topic

print("Topics ordered by per‑topic coherence (c_v), top words by FREX:\n")
for topic_id, coh in sorted_topics:
    # Get word indices sorted by smallest FREX score (best)
    best_word_ids = np.argsort(frex_score[topic_id, :])[:10]   # top 10
    words_list = [id2word[i] for i in best_word_ids]
    # Optionally add FREX values
    frex_vals = frex_score[topic_id, best_word_ids]
    parts = [f'{id2word[i]} (FREX={frex_score[topic_id,i]:.2f})' for i in best_word_ids]
    print(f"Topic {topic_id} (coherence={coh:.4f}): {', '.join(parts)}")
